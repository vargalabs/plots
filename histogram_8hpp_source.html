4 * plot::histogram(values, opts...) — a deferred view (plot::view) that bins a
5 * vector of doubles into N bins and draws count bars over a cartesian panel.
6 *
7 * Bin count: plot::bins{n} named argument; default ~Sturges' rule
8 * (ceil(log2(N)) + 1). Built on the #5 view pattern + impl::render_panel_into
9 * cartesian core: factory returns a movable struct tagged tag::view_t, captures
10 * data + options by value, exposes draw_into / natural / render / save.
11 * Dependency-free (standard library only).
12 */

38namespace plot::impl {
39 // resolve the bin count: plot::bins{n} override, else Sturges' rule.
40 template <class... opt_t>
41 std::size_t bins_of(std::size_t n, const opt_t&... opts){
42 using bins_t = typename arg::tpos<tag::bins_t, opt_t...>;
43 if constexpr( bins_t::present ){
44 auto tuple = std::forward_as_tuple(opts...);
45 std::size_t b = std::get<bins_t::position>(tuple).value;
46 return b ? b : 1;
47 }
48 if( n < 2 ) return 1;
49 return static_cast<std::size_t>(std::ceil(std::log2(double(n)))) + 1;
50 }

52 template <class... opt_t>
53 struct histogram_view {
54 using value_type = tag::view_t;
55 std::vector<double> values;
56 std::tuple<opt_t...> opts;
57
58 std::pair<std::size_t,std::size_t> natural() const {
59 return std::apply([](const auto&... o){
60 return impl::natural_size(std::size_t{640}, std::size_t{400}, o...); }, opts);
61 }
62 void draw_into(canvas_t& cv, float x, float y, float w, float h) const {
63 std::apply([&](const auto&... o){
64 const theme_t& th = impl::resolve_theme(o...);
65 auto [title, xl, yl] = impl::texts(o...);
66 const std::size_t nb = impl::bins_of(values.size(), o...);
67
68 auto [vmn, vmx] = impl::minmax_of(values);
69 if( vmx <= vmn ) vmx = vmn + 1.0;
70 const double width = (vmx - vmn) / double(nb);
71
72 std::vector<double> counts(nb, 0.0);
73 for(double v : values){
74 std::size_t b = static_cast<std::size_t>((v - vmn) / width);
75 if( b >= nb ) b = nb - 1;
76 counts[b] += 1.0;
77 }
78 double cmax = 0.0;
79 for(double c : counts) if( c > cmax ) cmax = c;
80 if( cmax <= 0.0 ) cmax = 1.0;
81
82 impl::scale_t sx = impl::make_scale(vmn, vmx, false, 10.0);
83 impl::scale_t sy = impl::make_scale(0.0, cmax, false, 10.0);
84 impl::render_panel_into(cv, x, y,
85 static_cast<std::size_t>(w), static_cast<std::size_t>(h),
86 th, sx, sy, title, xl, yl, {},
87 [&](impl::canvas_t& c, auto px, auto py){
88 using attribute_t = plot::attribute::element_t;
89 std::uint32_t col = th.series.empty()? th.fg : th.series[0];
90 const float y0 = py(0.0); // pixel baseline for count 0
91 for(std::size_t b=0;b<nb;++b){
92 double lo = vmn + double(b) * width;
93 double hi = vmn + double(b+1) * width;
94 float xl0 = px(lo), xr0 = px(hi);
95 float yt = py(counts[b]);
96 float bw = xr0 - xl0;
97 if( bw < 1.0f ) bw = 1.0f;
98 attribute_t a; a.color = plot::attribute::color_t{ col };
99 // inset by 1px so adjacent bars read as separate columns.
100 c.rect(xl0 + 0.5f, yt, bw - 1.0f, y0 - yt, 0, 0, a);
101 }
102 });
103 }, opts);
104 }
105 };
106}
