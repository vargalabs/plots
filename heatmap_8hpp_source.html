36 // Non-owning, row-major matrix view — the dependency-free replacement for the
37 // third-party dense matrix the original heatmap overload consumed.
38 template<class T> struct mat {
39 const T* p; std::size_t rows, cols;
40 T operator()(std::size_t i, std::size_t j) const { return p[i*cols + j]; }
41 const T* begin() const { return p; }
42 const T* end() const { return p + rows*cols; }
43 };
44}

56 // 3-stop blue -> yellow -> red linear gradient; t in [0,1] -> 0xRRGGBB.
57 inline std::uint32_t gradient(double t){
58 if( t < 0.0 ) t = 0.0; else if( t > 1.0 ) t = 1.0;
59 // stop colours
60 static constexpr double blue[3] = { 0.0, 64.0, 255.0};
61 static constexpr double yellow[3] = {255.0, 255.0, 0.0};
62 static constexpr double red[3] = {220.0, 30.0, 30.0};
63 double r,g,b;
64 if( t < 0.5 ){
65 double u = t / 0.5;
66 r = blue[0] + u*(yellow[0]-blue[0]);
67 g = blue[1] + u*(yellow[1]-blue[1]);
68 b = blue[2] + u*(yellow[2]-blue[2]);
69 } else {
70 double u = (t - 0.5) / 0.5;
71 r = yellow[0] + u*(red[0]-yellow[0]);
72 g = yellow[1] + u*(red[1]-yellow[1]);
73 b = yellow[2] + u*(red[2]-yellow[2]);
74 }
75 auto clamp8 = [](double v)->std::uint32_t {
76 if( v < 0 ) v = 0; else if( v > 255 ) v = 255;
77 return static_cast<std::uint32_t>(v + 0.5);
78 };
79 return (clamp8(r) << 16) | (clamp8(g) << 8) | clamp8(b);
80 }
81}

84 // ---- categorical heatmap over a mat<T> view (integral / enum) -------------
85 template <typename T>
86 typename std::enable_if<std::is_integral<T>::value || std::is_enum<T>::value>
87 ::type heatmap(impl::canvas_t& canvas, float x, float y,
88 std::vector<float>& dx, std::vector<float>& dy, float width, float height,
89 const plot::mat<T>& data, const plot::attribute::color_t& palette,
90 const std::array<std::uint32_t,3>& /*grad*/ ) {
91 using attribute_t = plot::attribute::element_t;
92 attribute_t mock_attr, gr_attr;
93 // find unique elements
94 std::vector<T> M(data.begin(), data.end());
95 std::sort(M.begin(), M.end());
96 auto last = std::unique(M.begin(), M.end());
97 M.erase(last, M.end());
98
99 canvas.group(static_cast<std::size_t>(x), static_cast<std::size_t>(y), mock_attr, [&]() -> void {
100 for( auto value : M ) {
101 gr_attr.color = palette[static_cast<std::ptrdiff_t>(value)];
102 canvas.group(std::size_t{0}, std::size_t{0}, gr_attr, [&]() -> void {
103 for(std::size_t j = 0; j < data.cols; j++) for( std::size_t i = 0; i < data.rows; i++)
104 if( data(i,j) == value )
105 canvas.rect( dx[j], dy[i], width, height, 1.5, 1.5, mock_attr);
106 });
107 }
108 });
109 }
110}

113 // ---- categorical heatmap over a point list (integral / enum) --------------
114 template <typename T>
115 typename std::enable_if<std::is_integral<T>::value || std::is_enum<T>::value>
116 ::type heatmap(impl::canvas_t& canvas, float x, float y,
117 std::vector<float>& dx, std::vector<float>& dy, float width, float height,
118 const std::vector<plot::data::point_t<T>>& data, const plot::attribute::color_t& palette,
119 const std::array<std::uint32_t,3>& /*grad*/ ) {
120 using attribute_t = plot::attribute::element_t;
121 attribute_t rect_attr, gr_attr;
122
123 gr_attr.color.reset();
124 canvas.group(static_cast<std::size_t>(x), static_cast<std::size_t>(y), gr_attr, [&]() -> void {
125 for(auto v : data){
126 rect_attr.href = v.href; rect_attr.label = v.label; rect_attr.color = palette[v.value];
127 canvas.rect(dx[v.x], dy[v.y], width, height, 1.5, 1.5, rect_attr);
128 }
129 });
130 }

134 // ---- continuous-gradient heatmap over a mat<T> view (arithmetic) ----------
135 // Each cell is normalized across the grid's [min,max] and coloured via the
136 // blue->yellow->red gradient; a <title> carries the numeric value.
137 template <typename T>
138 typename std::enable_if<std::is_arithmetic<T>::value && !std::is_integral<T>::value>
139 ::type heatmap(impl::canvas_t& canvas, float x, float y,
140 std::vector<float>& dx, std::vector<float>& dy, float width, float height,
141 const plot::mat<T>& data, const plot::attribute::color_t& /*palette*/,
142 const std::array<std::uint32_t,3>& grad ) {
143 using attribute_t = plot::attribute::element_t;
144 if( data.rows == 0 || data.cols == 0 ) return;
145
146 double lo = static_cast<double>(*data.begin());
147 double hi = lo;
148 for(const T& v : data){
149 double d = static_cast<double>(v);
150 if( d < lo ) lo = d;
151 if( d > hi ) hi = d;
152 }
153 const double span = (hi > lo) ? (hi - lo) : 1.0;
154
155 attribute_t gr_attr;
156 canvas.group(static_cast<std::size_t>(x), static_cast<std::size_t>(y), gr_attr, [&]() -> void {
157 for(std::size_t i = 0; i < data.rows; i++)
158 for(std::size_t j = 0; j < data.cols; j++){
159 double value = static_cast<double>(data(i,j));
160 double t = (value - lo) / span;
161 attribute_t cell;
162 cell.color = plot::attribute::color_t{ impl::gradient3(grad, t) };
163 cell.label = std::format("{:.4g}", value);
164 canvas.rect( dx[j], dy[i], width, height, 1.5, 1.5, cell );
165 }
166 });
167 }

171 // ---- the heatmap driver (writes a standalone .svg to an ostream) ----------
172 template <class T, class... arg_t>
173 void heatmap(std::ostream& os, const T& data, arg_t... args ) {
174 using title_t = typename arg::tpos<tag::title_t, arg_t...>;
175 using footnote_t = typename arg::tpos<tag::footnote_t, arg_t...>;
176 using legend_t = typename arg::tpos<tag::legend_t, arg_t...>;
177 using axisx_t = typename arg::tpos<tag::axis::x_t, arg_t...>;
178 using axisy_t = typename arg::tpos<tag::axis::y_t, arg_t...>;
179 using width_t = typename arg::tpos<tag::width_t, arg_t...>;
180 using height_t = typename arg::tpos<tag::height_t, arg_t...>;
181 using margin_t = typename arg::tpos<tag::margin_t, arg_t...>;
182
183 static_assert( axisx_t::present, "x axis must be specified..." );
184 static_assert( axisy_t::present, "y axis must be specified..." );
185
186 auto tuple = std::forward_as_tuple(args...);
187
188 std::array<float,4> margin{5,5,5,5};
189 if constexpr( margin_t::present )
190 margin = std::get<margin_t::position>( tuple ).value;
191
192 auto x_axis = std::get<axisx_t::position>( tuple );
193 auto y_axis = std::get<axisy_t::position>( tuple );
194
195 // when a title is present, push the x-axis (and thus the grid below it)
196 // down by a band so the title sits above the ticks instead of overlapping
197 // the top of the x-axis. The canvas auto-grows in height to match.
198 if constexpr( title_t::present ){
199 x_axis.position = position{ std::get<std::size_t>(x_axis.position->x),
200 std::get<std::size_t>(x_axis.position->y) + std::size_t{14} };
201 }
202
203 float offset_x = std::get<std::size_t>(x_axis.position->x) - .5f * x_axis.grid;
204 float offset_y = std::get<std::size_t>(x_axis.position->y) + .5f * y_axis.grid;
205 // when y axis label position is not preset, compute it from x layout
206 float pos_x = x_axis.dx.back() + 1.5f * x_axis.grid + offset_x;
207 y_axis.position = position{static_cast<std::size_t>(pos_x), static_cast<std::size_t>(offset_y + .5f * y_axis.grid) };
208 // compute size when not set
209 std::size_t width, height;
210 if constexpr( width_t::present )
211 width = std::get<width_t::position>( tuple ).value;
212 else
213 width = static_cast<std::size_t>(y_axis.get_x() + margin[2]);
214 if constexpr( height_t::present )
215 height = std::get<height_t::position>( tuple ).value;
216 else
217 height = static_cast<std::size_t>(y_axis.get_y() + offset_y + margin[3]);
218
219 const theme_t& th = impl::resolve_theme(args...);
220
221 auto canvas = plot::impl::canvas_t(os, width, height + 10, margin);
222 // theme background spanning the whole canvas (painted first, under all).
223 { plot::attribute::element_t bg; bg.color = plot::attribute::color_t{ th.bg };
224 canvas.rect(0, 0, static_cast<float>(width), static_cast<float>(height + 10), 0, 0, bg); }
225 canvas << x_axis; canvas << y_axis;
226
227 if constexpr (title_t::present) canvas << std::get<title_t::position>( tuple );
228 if constexpr (footnote_t::present) canvas << std::get<footnote_t::position>( tuple );
229
230 plot::attribute::color_t palette{ 0x4060FF };
231 if constexpr (legend_t::present) {
232 canvas << std::get<legend_t::position>( tuple );
233 auto legend = std::get<legend_t::position>( tuple );
234 if( legend.color ) palette = legend.color.value();
235 }
236 impl::heatmap(canvas,
237 offset_x, offset_y, x_axis.dx, y_axis.dy,
238 .9f * x_axis.grid, .9f * y_axis.grid, data, palette, th.gradient );
239 }

241 // filename convenience: opens a truncating ofstream and renders into it.
242 template <class T, class... arg_t>
243 void heatmap(const std::string& filename, const T& data, arg_t... args ) {
244 std::ofstream ofs(filename, std::ios::out | std::ios::trunc);
245 heatmap(ofs, data, args...);
246 }
