4 * plot::pie(labels, values, opts...) — a deferred view (plot::view) drawing a
5 * pie chart, or a donut when plot::donut{r} (inner-radius fraction in (0,1)) is
6 * given. Each slice is a FILLED <polygon> wedge: the arc is segmented into short
7 * chords, centre→arc→centre for a pie, or outer-arc→inner-arc for a donut ring.
8 * Slices are filled with the theme colour. Slice colours cycle theme.series; a
9 * label is placed at the slice's mid-angle.
10 *
11 * Built on the #5 view pattern (draws straight into the cell). Dependency-free.
12 */

33namespace plot {
34 // donut inner-radius fraction in (0,1); 0 (or absent) ⇒ a full pie.
35 struct donut { using value_type = tag::donut_t; double value; };
36}
37
38namespace plot::impl {
39 template <class... opt_t>
40 double donut_of(const opt_t&... opts){
41 using d_t = typename arg::tpos<tag::donut_t, opt_t...>;
42 if constexpr( d_t::present ){
43 auto tuple = std::forward_as_tuple(opts...);
44 double r = std::get<d_t::position>(tuple).value;
45 if( r < 0 ) r = 0; else if( r > 0.95 ) r = 0.95;
46 return r;
47 }
48 return 0.0;
49 }
50
51 template <class... opt_t>
52 struct pie_view {
53 using value_type = tag::view_t;
54 std::vector<std::string> labels;
55 std::vector<double> values;
56 std::tuple<opt_t...> opts;
57
58 std::pair<std::size_t,std::size_t> natural() const {
59 return std::apply([](const auto&... o){
60 return impl::natural_size(std::size_t{500}, std::size_t{500}, o...); }, opts);
61 }
62 void draw_into(canvas_t& cv, float x, float y, float w, float h) const {
63 std::apply([&](const auto&... o){
64 const theme_t& th = impl::resolve_theme(o...);
65 auto [title, xl, yl] = impl::texts(o...);
66 const double inner = impl::donut_of(o...);
67 using attribute_t = plot::attribute::element_t;
68
69 const std::size_t n = std::min(labels.size(), values.size());
70 double total = 0.0; for(std::size_t i=0;i<n;++i) if(values[i]>0) total += values[i];
71 if( total <= 0.0 ) total = 1.0;
72
73 cv.group(static_cast<std::size_t>(x), static_cast<std::size_t>(y), attribute_t{},
74 [&](){
75 { attribute_t bg; bg.color = plot::attribute::color_t{ th.bg };
76 cv.rect(0,0, w, h, 0,0, bg); }
77 float top = title.empty() ? 8.0f : 26.0f;
78 if( !title.empty() ){
79 attribute_t a; a.color = plot::attribute::color_t{ th.fg };
80 a.font = plot::attribute::font_t{"Arial, sans-serif", "bold", 13u};
81 a.align = plot::attribute::align_t::center;
82 cv.text(title, std::size_t(w/2), std::size_t(16), a);
83 }
84 float cx = w*0.5f, cy = top + (h - top)*0.5f;
85 float Ro = std::max(8.0f, std::min(w, h - top)*0.5f - 30.0f);
86 float Ri = float(inner) * Ro;
87
88 double a0 = -1.5707963267948966; // start at 12 o'clock
89 const double TWO_PI = 6.283185307179586;
90 for(std::size_t i=0;i<n;++i){
91 double frac = (values[i] > 0 ? values[i] : 0.0) / total;
92 double a1 = a0 + frac*TWO_PI;
93 // segment the arc into ~enough chords for smoothness.
94 std::size_t segs = std::max<std::size_t>(2,
95 static_cast<std::size_t>(std::ceil(frac*64.0)));
96 std::vector<float> X, Y;
97 // outer arc a0 -> a1
98 for(std::size_t s=0;s<=segs;++s){
99 double a = a0 + (a1-a0)*double(s)/double(segs);
100 X.push_back(cx + Ro*float(std::cos(a)));
101 Y.push_back(cy + Ro*float(std::sin(a)));
102 }
103 if( Ri > 0.5f ){
104 // inner arc a1 -> a0 (reverse) for a ring wedge.
105 for(std::size_t s=0;s<=segs;++s){
106 double a = a1 + (a0-a1)*double(s)/double(segs);
107 X.push_back(cx + Ri*float(std::cos(a)));
108 Y.push_back(cy + Ri*float(std::sin(a)));
109 }
110 } else {
111 X.push_back(cx); Y.push_back(cy); // pie apex
112 }
113 // close the wedge.
114 X.push_back(X.front()); Y.push_back(Y.front());
115
116 std::uint32_t col = th.series.empty()? th.fg
117 : th.series[i % th.series.size()];
118 attribute_t pa; pa.color = plot::attribute::color_t{ col };
119 // subtle separator stroke in the background colour so
120 // adjacent filled slices read as distinct wedges.
121 pa.stroke = plot::attribute::stroke_t{1.0f, 1.0f, {}, {}, {}};
122 cv.polygon(X, Y, pa);
123
124 // label at the slice mid-angle, just outside the wedge.
125 double am = 0.5*(a0+a1);
126 float lr = Ro + 12.0f;
127 attribute_t ta; ta.color = plot::attribute::color_t{ th.fg };
128 ta.font = plot::attribute::font_t{"Arial, sans-serif", "normal", 10u};
129 ta.align = plot::attribute::align_t::center;
130 float lx = cx + lr*float(std::cos(am));
131 float ly = cy + lr*float(std::sin(am));
132 cv.text(labels[i], std::size_t(lx<0?0:lx), std::size_t(ly+3), ta);
133 a0 = a1;
134 }
135 });
136 }, opts);
137 }
138 };
139}
